variable "vm-name" {
    type = string
    default = "vm-name-"
    description = "vm name"
  
}

variable "vm-count" {
    default = 3
    type = string
    description = "vm count"
  
}


resource "azurerm_resource_group" "rg01" {

    name = "rg"
  location = "south india"
  
}


resource "azurerm_virtual_network" "vnet" {
    name = "vnet1"
    location = azurerm_resource_group.rg01.location
  resource_group_name = azurerm_resource_group.rg01.name
  address_space = ["10.0.0.0/16"]
}

resource "azurerm_subnet" "subnet" {
    name = "subnet"
  virtual_network_name = azurerm_virtual_network.vnet.name
  resource_group_name = azurerm_resource_group.rg01.name
  address_prefixes = ["10.0.1.0/24"]

}


resource "azurerm_public_ip" "pub" {
    count = var.vm-count
    name = "${var.vm-name}-${count.index}-pip"
    location = azurerm_resource_group.rg01.location
  resource_group_name = azurerm_resource_group.rg01.name
  allocation_method = "Static"

}

resource "azurerm_network_interface" "nic" {
    count = var.vm-count
    name = "${var.vm-name}-${count.index}-nic"
  location = azurerm_resource_group.rg01.location
  resource_group_name = azurerm_resource_group.rg01.name

  ip_configuration {
    name            = "internal"
    subnet_id = azurerm_subnet.subnet.id
    private_ip_address_allocation = "Dynamic"
    public_ip_address_id = azurerm_public_ip.pub[count.index].id
    
    
    }
}    
resource "azurerm_windows_virtual_machine" "example" {
    count = var.vm-count
  name                = "${var.vm-name}-${count.index}-vm"
  resource_group_name = azurerm_resource_group.rg01.name
  location            = azurerm_resource_group.rg01.location
  size                = "Standard_B2ls_v2"
  admin_username      = "azadmin"
  admin_password      = "auto@123"
  network_interface_ids = [
    azurerm_network_interface.nic[count.index].id,
  ]

  os_disk {
    caching              = "ReadWrite"
    storage_account_type = "Standard_LRS"
  }

  source_image_reference {
    publisher = "MicrosoftWindowsServer"
    offer     = "WindowsServer"
    sku       = "2016-Datacenter"
    version   = "latest"
  }
}
